// 轨迹规划基础
// 学习重点：先给出起终位置、速度和加速度要求。
// 选择一条满足边界条件的平滑位移曲线。
// 求速度与加速度，检查峰值是否超过约束。
// C++17；直接编译运行 main() 中的示例。
#include <iostream>  // 控制台输入与输出
#include <algorithm>  // 排序、取最值、合并或反转等标准算法
#include <cmath>  // 距离、幂、平方根或数值判断
#include <iomanip>  // 控制输出精度与格式
#include <stdexcept>  // 明确报告非法输入
#include <array>  // 固定长度坐标或障碍边界

using namespace std;

// 按物理时刻求五次位移曲线及其一、二阶时间导数。
array<double,3> trajectory(double displacement, double duration, double time) {
    // 不符合本算法前提的输入要明确拒绝，不能带着错误条件继续计算。
    if (duration <= 0 || time < 0 || time > duration) throw invalid_argument("invalid time");
    // 把物理时间转换为0到1的归一化时间；求导后仍需除以实际时长。
    double tau = time/duration;
    double s = displacement*(10*pow(tau,3)-15*pow(tau,4)+6*pow(tau,5));
    // 一次时间导数带1/T，得到米每秒。
    double v = displacement/duration*(30*pow(tau,2)-60*pow(tau,3)+30*pow(tau,4));
    // 二次时间导数带1/T²，得到米每二次方秒。
    double a = displacement/(duration*duration)*(60*tau-180*pow(tau,2)+120*pow(tau,3));
    return {s,v,a};
}

// 程序入口：准备示例数据，调用算法，并输出结果供手算核对。
int main() {
    double displacement = 10, maxVelocity = 2, maxAcceleration = 1;
    double duration = max(15.0/8*abs(displacement)/maxVelocity,
                          sqrt(10*sqrt(3.0)/3*abs(displacement)/maxAcceleration));
    // 输出计算结果，便于与课程例子或自行推演的答案核对。
    cout << fixed << setprecision(6) << "T=" << duration << '\n';
    cout << "t,s,v,a\n";
    for (int i = 0; i <= 40; ++i) {
        double time = duration*i/40;
        auto [s,v,a] = trajectory(displacement,duration,time);
        cout << time << ',' << s << ',' << v << ',' << a << '\n';
    }
}

# 轨迹规划基础
# Python 3；直接运行文件，修改示例数据可继续练习。
# 先给出起终位置、速度和加速度要求。

import math

# 对满足静止边界条件的五次多项式采样，返回时间、位置、速度、加速度。
def quintic_trajectory(distance, duration, steps=40):
    if duration <= 0 or steps < 1:
        # 不符合本算法前提的输入要明确拒绝，不能带着错误条件继续计算。
        raise ValueError('总时间和采样步数必须为正')
    result = []
    for i in range(steps + 1):
        # 把物理时间转换为0到1的归一化时间；求导后仍需除以实际时长。
        tau = i / steps
        position = distance * (10 * tau ** 3 - 15 * tau ** 4 + 6 * tau ** 5)
        # 一次时间导数带1/T，得到米每秒。
        velocity = distance / duration * (30 * tau ** 2 - 60 * tau ** 3 + 30 * tau ** 4)
        # 二次时间导数带1/T²，得到米每二次方秒。
        acceleration = distance / duration ** 2 * (60 * tau - 180 * tau ** 2 + 120 * tau ** 3)
        result.append((duration * tau, position, velocity, acceleration))
    return result

# 程序入口：准备示例数据，调用算法，并输出结果供手算核对。
def main():
    # 二次时间导数带1/T²，得到米每二次方秒。
    distance, max_velocity, max_acceleration = (10, 2, 1)
    duration = max(15 / 8 * abs(distance) / max_velocity, math.sqrt(10 * math.sqrt(3) / 3 * abs(distance) / max_acceleration))
    # 输出计算结果，便于与课程例子或自行推演的答案核对。
    print('T:', duration)
    print('t,s,v,a')
    for row in quintic_trajectory(distance, duration):
        print(','.join((f'{value:.6f}' for value in row)))
if __name__ == '__main__':
    main()

写代码时最容易错的地方

归一化时间导数未换算

s(τ)=10τ 米，$\tau=t/T$，时长 T=5 秒。

容易写错
velocity = ds_dtau;
修改后
velocity = ds_dtau / duration;

链式法则给出 ds/dt=(ds/dτ)/T，速度单位是米/秒。
