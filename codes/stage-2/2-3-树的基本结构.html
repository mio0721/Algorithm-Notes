C++ 完整示例

下载 2-3-树的基本结构.cpp

#include <iostream>
#include <vector>
#include <queue>
#include <stack>
#include <unordered_map>
#include <functional>
#include <algorithm>
using namespace std;

struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
};

int main() {
    TreeNode a(1), b(2), c(3), d(4);
    a.left=&b; a.right=&c; b.left=&d;
    cout << "根: " << a.val << "，左子节点: " << a.left->val << "，左孙节点: " << a.left->left->val << "\n";
    return 0;
}

Python 完整示例

下载 2-3-树的基本结构.py

class TreeNode:
    def __init__(self, value):
        self.val = value
        self.left = None
        self.right = None

root = TreeNode(1)
root.left = TreeNode(2)
root.right = TreeNode(3)
root.left.left = TreeNode(4)
print("根:", root.val, "左子节点:", root.left.val, "左孙节点:", root.left.left.val)
