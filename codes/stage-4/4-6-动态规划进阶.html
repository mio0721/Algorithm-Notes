// 动态规划进阶
// 学习重点：先考虑两份记录很短的前缀。
// 末尾相同，就把这一对字符接在较短问题的结果后面。
// 末尾不同，就分别尝试跳过其中一边的末尾。
// C++17；直接编译运行 main() 中的示例。
#include <iostream>  // 控制台输入与输出
#include <vector>  // 动态数组、邻接表或状态表
#include <string>  // 字符串存储与操作
#include <algorithm>  // 排序、取最值、合并或反转等标准算法

using namespace std;

// 填公共前缀状态表，再沿最优邻格回溯出一个最长公共子序列。
string lcs(const string& a, const string& b) {
    int m = static_cast<int>(a.size()), n = static_cast<int>(b.size());
    vector<vector<int>> dp(m + 1, vector<int>(n + 1));
    // 第0行、0列表示空前缀，所以真实字符从下标i−1、j−1读取。
    for (int i = 1; i <= m; ++i) {
        for (int j = 1; j <= n; ++j) {
            dp[i][j] = a[i - 1] == b[j - 1]
                ? dp[i - 1][j - 1] + 1 : max(dp[i - 1][j], dp[i][j - 1]);
        }
    }
    string result;
    while (m > 0 && n > 0) {
        if (a[m - 1] == b[n - 1]) {
            result.push_back(a[--m]);
            --n;
        } else if (dp[m - 1][n] >= dp[m][n - 1]) {
            --m;
        } else {
            --n;
        }
    }
    // 回溯收集的是倒序公共字符，输出前需要反转。
    reverse(result.begin(), result.end());
    return result;
}

// 程序入口：准备示例数据，调用算法，并输出结果供手算核对。
int main() {
    string result = lcs("ABCBDAB", "BDCABA");
    // 输出计算结果，便于与课程例子或自行推演的答案核对。
    cout << result.size() << ' ' << result << '\n';
}

# 动态规划进阶
# Python 3；直接运行文件，修改示例数据可继续练习。
# 先考虑两份记录很短的前缀。

# 填公共前缀状态表，再沿最优邻格回溯出一个最长公共子序列。
def lcs(first, second):
    m, n = (len(first), len(second))
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    # 第0行、0列表示空前缀，所以真实字符从下标i−1、j−1读取。
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if first[i - 1] == second[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    path, i, j = ([], m, n)
    while i and j:
        if first[i - 1] == second[j - 1]:
            path.append(first[i - 1])
            i, j = (i - 1, j - 1)
        elif dp[i - 1][j] >= dp[i][j - 1]:
            i -= 1
        else:
            j -= 1
    # 回溯收集的是倒序公共字符，输出前需要反转。
    return (dp[m][n], ''.join(reversed(path)))

# 程序入口：准备示例数据，调用算法，并输出结果供手算核对。
def main():
    # 输出计算结果，便于与课程例子或自行推演的答案核对。
    print(lcs('ABCBDAB', 'BDCABA'))
if __name__ == '__main__':
    main()

写代码时最容易错的地方

子序列误当连续子串

比较 abc 与 ac。

容易写错
// 因为 a 和 c 不相邻而丢弃
修改后
// 允许跳过字符，但保留顺序

ac 在 abc 中不连续，仍是合法子序列。

前缀长度直接当字符下标

前缀长度 i=1，字符串为 a。

容易写错
if (s[i] == t[j]) dp[i][j]=...;
修改后
if (s[i-1] == t[j-1]) dp[i][j]=...;

前缀长度 1 的末字符在下标 0；C++ s[size()] 读到终止字符，不能当作有效字符。

运行后怎样检查？

字符不同不能取左上角+1；回溯得到的是倒序字符，要反转。有多个最优解时输出任一个即可，不要要求字符串唯一。
