// Floyd–Warshall
// 学习重点：先填好直接边的距离表。
// 选一个中转点，查看任意起终点经它是否更近。
// 所有中转点都试完，得到任意两点的最短距离。
// C++17；直接编译运行 main() 中的示例。
#include <iostream>  // 控制台输入与输出
#include <vector>  // 动态数组、邻接表或状态表
#include <algorithm>  // 排序、取最值、合并或反转等标准算法
#include <limits>  // 整数边界或浮点无穷大
#include <stdexcept>  // 明确报告非法输入

using namespace std;

const long long INF = numeric_limits<long long>::max()/4;
// 逐轮开放中转点，只有两段均可达时才尝试更新距离。
vector<vector<long long>> floyd(vector<vector<long long>> distance) {
    int n = static_cast<int>(distance.size());
    // 不符合本算法前提的输入要明确拒绝，不能带着错误条件继续计算。
    for (const auto& row : distance) if (static_cast<int>(row.size()) != n) throw invalid_argument("not square");
    // 中转点必须在最外层，表示本轮新允许使用哪个中转点。
    for (int k = 0; k < n; ++k) {
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < n; ++j) {
                // 两段都可达时才相加，避免把无穷大标记参与整数运算。
                if (distance[i][k] != INF && distance[k][j] != INF) {
                    distance[i][j] = min(distance[i][j], distance[i][k]+distance[k][j]);
                }
            }
        }
    }
    // 不符合本算法前提的输入要明确拒绝，不能带着错误条件继续计算。
    for (int i = 0; i < n; ++i) if (distance[i][i] < 0) throw invalid_argument("negative cycle");
    return distance;
}

// 程序入口：准备示例数据，调用算法，并输出结果供手算核对。
int main() {
    vector<vector<long long>> distance{{0,3,10,INF},{INF,0,4,8},{INF,INF,0,2},{1,INF,INF,0}};
    for (const auto& row : floyd(distance)) {
        for (long long cost : row) {
            // 输出计算结果，便于与课程例子或自行推演的答案核对。
            if (cost == INF) cout << "INF "; else cout << cost << ' ';
        }
        cout << '\n';
    }
}

# Floyd–Warshall
# Python 3；直接运行文件，修改示例数据可继续练习。
# 先填好直接边的距离表。

import math

# 逐轮开放中转点，计算全点对最短路，并检查是否存在负环。
def floyd_warshall(matrix):
    d = [row.copy() for row in matrix]
    # 中转点必须在最外层，表示本轮新允许使用哪个中转点。
    for k in range(len(d)):
        for i in range(len(d)):
            for j in range(len(d)):
                d[i][j] = min(d[i][j], d[i][k] + d[k][j])
    if any((d[i][i] < 0 for i in range(len(d)))):
        # 不符合本算法前提的输入要明确拒绝，不能带着错误条件继续计算。
        raise ValueError('检测到负环，不能直接把矩阵解释为有限最短距离')
    return d

# 程序入口：准备示例数据，调用算法，并输出结果供手算核对。
def main():
    inf = math.inf
    matrix = [[0, 3, 10, inf], [inf, 0, 4, 8], [inf, inf, 0, 2], [1, inf, inf, 0]]
    for row in floyd_warshall(matrix):
        # 输出计算结果，便于与课程例子或自行推演的答案核对。
        print(row)
if __name__ == '__main__':
    main()

写代码时最容易错的地方

中转点循环放在内层

有向路径 0→3→2→1，各边权重均为 1。

容易写错
for (int i=0;i<n;++i)
 for (int j=0;j<n;++j)
  for (int k=0;k<n;++k) relax(i,j,k);
修改后
for (int k=0;k<n;++k)
 for (int i=0;i<n;++i)
  for (int j=0;j<n;++j) relax(i,j,k);

i、j 固定时，0→2 尚未更新就检查了 0→1，单轮可能漏掉路径。

无穷大参与加法溢出

用 INT_MAX 表示不可达，候选边权为 5。

容易写错
dist[i][j]=min(dist[i][j],dist[i][k]+dist[k][j]);
修改后
if (dist[i][k]!=INF && dist[k][j]!=INF)
    dist[i][j]=min(dist[i][j],dist[i][k]+dist[k][j]);

不可达标记不能直接相加；有限距离之和也要选择足够宽的类型。

运行后怎样检查？

k必须在最外层。没有边的距离是∞，自身是0；C++相加前判断可达性避免整型溢出。多条平行边初始化取最小权。
