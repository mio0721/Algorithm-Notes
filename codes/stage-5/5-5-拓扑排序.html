// 拓扑排序
// 学习重点：数清每个任务还有几个未完成的前置任务。
// 把没有前置阻碍的任务加入队列。
// 做完一个任务，减少它后续任务的剩余依赖。
// C++17；直接编译运行 main() 中的示例。
#include <iostream>  // 控制台输入与输出
#include <vector>  // 动态数组、邻接表或状态表
#include <queue>  // 先进先出队列或优先队列
#include <stdexcept>  // 明确报告非法输入

using namespace std;

// 持续处理没有剩余前置依赖的任务；处理不完全部任务就报告有向环。
vector<int> topologicalSort(const vector<vector<int>>& graph) {
    vector<int> indegree(graph.size());
    for (const auto& edges : graph) for (int v : edges) ++indegree[v];
    queue<int> ready;
    for (int u = 0; u < static_cast<int>(graph.size()); ++u) if (!indegree[u]) ready.push(u);
    vector<int> order;
    // 反复取出一个待处理状态；集合耗尽时，本轮搜索结束。
    while (!ready.empty()) {
        // 读取先进先出的队首，保持发现的层次顺序。
        int u = ready.front();
        ready.pop();
        order.push_back(u);
        // 完成当前任务后，后继少一个未满足依赖；恰好减到0才入队。
        for (int v : graph[u]) if (--indegree[v] == 0) ready.push(v);
    }
    // 未能处理全部顶点意味着存在依赖环，不能把部分顺序当完整答案。
    if (order.size() != graph.size()) throw invalid_argument("dependency cycle");
    return order;
}

// 程序入口：准备示例数据，调用算法，并输出结果供手算核对。
int main() {
    vector<vector<int>> graph{{2,3},{3},{4},{4,5},{6},{7},{8},{8},{9},{}};
    // 输出计算结果，便于与课程例子或自行推演的答案核对。
    for (int u : topologicalSort(graph)) cout << char('A'+u) << ' ';
    cout << '\n';
    try {
        topologicalSort({{1},{0}});
    } catch (const invalid_argument& error) {
        cout << error.what() << '\n';
    }
}

# 拓扑排序
# Python 3；直接运行文件，修改示例数据可继续练习。
# 数清每个任务还有几个未完成的前置任务。

from collections import deque

# 持续处理没有剩余前置依赖的任务；处理不完全部任务就报告有向环。
def topological_sort(graph):
    degree = [0] * len(graph)
    for row in graph:
        for v in row:
            degree[v] += 1
    pending = deque((i for i, d in enumerate(degree) if d == 0))
    order = []
    # 反复取出一个待处理状态；集合耗尽时，本轮搜索结束。
    while pending:
        # 读取先进先出的队首，保持发现的层次顺序。
        u = pending.popleft()
        order.append(u)
        for v in graph[u]:
            degree[v] -= 1
            if degree[v] == 0:
                pending.append(v)
    # 未能处理全部顶点意味着存在依赖环，不能把部分顺序当完整答案。
    if len(order) != len(graph):
        # 不符合本算法前提的输入要明确拒绝，不能带着错误条件继续计算。
        raise ValueError('图中有环，无法完成拓扑排序')
    return order

# 程序入口：准备示例数据，调用算法，并输出结果供手算核对。
def main():
    graph = [[2, 3], [3], [4], [4, 5], [6], [7], [8], [8], [9], []]
    # 输出计算结果，便于与课程例子或自行推演的答案核对。
    print(topological_sort(graph))
    try:
        topological_sort([[1], [0]])
    except ValueError as error:
        print(error)
if __name__ == '__main__':
    main()

依赖方向读反

容易写错

修改后
graph[b].push_back(a);

边应从前置任务指向后续任务。

运行后怎样检查？

拓扑序不是BFS的层序，也不一定唯一。多个0入度顶点可以按不同顺序处理；验证应检查每条依赖边的先后关系。
